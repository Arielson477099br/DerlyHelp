    function loadJson(key, fallback) {
      try {
        const value = localStorage.getItem(key);
        return value ? JSON.parse(value) : fallback;
      } catch {
        return fallback;
      }
    }

    function saveJson(key, value) {
      localStorage.setItem(key, JSON.stringify(value));
    }

    function sanitizeText(value) {
      return String(value ?? "")
        .replace(/\s+/g, " ")
        .trim();
    }

    function escapeHtml(value) {
      return String(value ?? "")
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function normalizePhone(value) {
      const raw = String(value ?? "").trim();
      const digits = raw.replace(/\D/g, "");
      return digits ? `tel:${digits}` : "#";
    }

    function initials(name) {
      const parts = sanitizeText(name).split(" ").filter(Boolean);
      if (!parts.length) return "?";
      return parts.slice(0, 2).map(p => p[0].toUpperCase()).join("");
    }

    function speechAvailable() {
      return "speechSynthesis" in window;
    }

    function getPortugueseVoice() {
      if (!speechAvailable()) return null;
      const voices = window.speechSynthesis.getVoices();
      return (
        voices.find(v => /^pt-BR$/i.test(v.lang)) ||
        voices.find(v => /^pt/i.test(v.lang)) ||
        voices[0] ||
        null
      );
    }

    function speak(text, options = {}) {
      if (!speechAvailable()) {
        showToast("A fala não está disponível", "O navegador deste aparelho não oferece leitura em voz alta.");
        return;
      }

      const clean = sanitizeText(text);
      if (!clean) return;

      window.speechSynthesis.cancel();

      const utterance = new SpeechSynthesisUtterance(clean);
      utterance.lang = "pt-BR";
      utterance.rate = options.rate || 0.9;
      utterance.pitch = 1;
      utterance.volume = 1;

      const voice = getPortugueseVoice();
      if (voice) utterance.voice = voice;

      utterance.onstart = () => {
        els.speakPageLabel.textContent = "Parar de ouvir";
      };

      utterance.onend = () => {
        els.speakPageLabel.textContent = "Ouvir página";
      };

      utterance.onerror = () => {
        els.speakPageLabel.textContent = "Ouvir página";
      };

      window.speechSynthesis.speak(utterance);
    }

    }

        els.callDialog.showModal();
      } else {
        const ok = window.confirm(text);
        if (ok) performCall(contact);
      }

      speak(text);
    }

    function performCall(contact) {
      if (!contact) return;

      const spoken = `Ligando para ${contact.name}.`;
      speak(spoken);

      showToast("Ligação iniciada", `Você escolheu ligar para ${contact.name}.`, false);

      const href = normalizePhone(contact.phone);
      if (href !== "#") {
        window.location.href = href;
      }

      pendingCall = null;
    }

    function renderMedications(query = "") {
      const q = sanitizeText(query).toLowerCase();

      const list = otcExamples.filter(m =>
        !q ||
        m.name.toLowerCase().includes(q) ||
        m.symptoms.toLowerCase().includes(q)
      );

      if (!list.length) {
        els.medList.innerHTML = `<div class="empty-state">Nenhum exemplo encontrado. Tente outro nome ou sintoma.</div>`;
        return;
      }

      els.medList.innerHTML = list.map(m => `
        <article class="med-card">
          <h4>${escapeHtml(m.name)}</h4>
          <p><strong>Relacionado a:</strong> ${escapeHtml(m.symptoms)}</p>
          <p>${escapeHtml(m.note)}</p>
          <button type="button" class="btn-small" data-action="speak-med"
            data-name="${escapeHtml(m.name)}"
            data-symptoms="${escapeHtml(m.symptoms)}">
            <span class="icon" aria-hidden="true">▶</span> Ouvir
          </button>
        </article>
      `).join("");

    function addMedReminder() {
      const name = sanitizeText(els.medName.value);
      const time = sanitizeText(els.medTime.value);
      const note = sanitizeText(els.medNote.value);

      if (!name || !time) {
        showToast("Falta uma informação", "Preencha o nome e o horário.");
        return;
      }

      state.medReminders.push({
        id: crypto.randomUUID ? crypto.randomUUID() : String(Date.now()),
        name,
        time,
        note
      });

      saveJson(STORAGE.medReminders, state.medReminders);
      els.medName.value = "";
      els.medTime.value = "";
      els.medNote.value = "";
      renderMedReminders();
      showToast("Horário salvo", `Lembrete para ${name} às ${time}.`);
    }

    function deleteMedReminder(id) {
      const item = state.medReminders.find(m => m.id === id);
      state.medReminders = state.medReminders.filter(m => m.id !== id);
      saveJson(STORAGE.medReminders, state.medReminders);
      renderMedReminders();
      if (item) showToast("Lembrete apagado", `O horário de ${item.name} foi removido.`);
    }

    function renderMedReminders() {
      if (!state.medReminders.length) {
        els.medReminderList.innerHTML = `<div class="empty-state">Nenhum horário anotado.</div>`;
        updateSummary();
        return;
      }

      const sorted = [...state.medReminders].sort((a, b) => a.time.localeCompare(b.time));

      els.medReminderList.innerHTML = sorted.map(item => `
        <article class="task-item">
          <div class="check-btn" aria-hidden="true">◷</div>
          <div>
            <strong>${escapeHtml(item.name)}</strong>
            <div class="task-meta">Horário: ${escapeHtml(item.time)}${item.note ? ` | ${escapeHtml(item.note)}` : ""}</div>
          </div>
          <button type="button" class="task-remove btn-small btn-danger"
            data-action="delete-med-reminder"
            data-id="${escapeHtml(item.id)}">Apagar</button>
        </article>
      `).join("");

      updateSummary();
    }

    function updateSummary() {
      const pending = state.tasks.filter(t => !t.completed).length;
      els.summaryTasks.textContent = String(pending);

      const reminders = [
        ...state.medReminders.map(r => ({ time: r.time, label: r.name })),
        ...state.tasks
          .filter(t => !t.completed && t.time)
          .map(t => ({ time: t.time, label: t.text }))
      ].sort((a, b) => a.time.localeCompare(b.time));

      if (reminders.length) {
        els.summaryNextReminder.textContent = reminders[0].time;
        els.summaryReminderDetails.textContent = reminders[0].label;
      } else {
        els.summaryNextReminder.textContent = "Nenhum";
        els.summaryReminderDetails.textContent = "Você ainda não registrou um horário.";
      }

      const main = state.contacts[0];
      els.summaryContact.textContent = main ? main.name : "Não definido";
    }

    /* -------------------------
       LEMBRETES SONOROS E VISUAIS
       ------------------------- */
    function beep() {
      try {
        audioContext = audioContext || new (window.AudioContext || window.webkitAudioContext)();
        const oscillator = audioContext.createOscillator();
        const gain = audioContext.createGain();

        oscillator.type = "sine";
        oscillator.frequency.value = 880;
        gain.gain.value = 0.05;

        oscillator.connect(gain);
        gain.connect(audioContext.destination);

        oscillator.start();
        oscillator.stop(audioContext.currentTime + 0.35);
      } catch {
        // Alguns navegadores bloqueiam áudio até uma interação. A fala ainda pode funcionar.
      }
    }

    function reminderKey(kind, id, date) {
      return `${kind}:${id}:${date}`;
    }

    function checkReminders(now) {
      const hhmm = now.toTimeString().slice(0, 5);
      const yyyyMMdd = now.toISOString().slice(0, 10);

      let changed = false;
      for (const task of state.tasks) {
        if (
          !task.completed &&
          task.time &&
          task.time === hhmm &&
          (!task.date || task.date === yyyyMMdd)
        ) {
          const key = reminderKey("task", task.id, yyyyMMdd);
          if (!state.remindersDone[key]) {
            triggerReminder(`Lembrete de tarefa`, task.text, key);
            changed = true;
          }
        }
      }

      for (const med of state.medReminders) {
        if (med.time === hhmm) {
          const key = reminderKey("med", med.id, yyyyMMdd);
          if (!state.remindersDone[key]) {
            triggerReminder(`Lembrete de horário`, med.name, key, med.note);
            changed = true;
          }
        }
      }

      if (changed) {
        saveJson(STORAGE.remindersDone, state.remindersDone);
      }

      /* Mantém o objeto leve: remove registros com mais de 14 dias. */
      const threshold = new Date(now);
      threshold.setDate(threshold.getDate() - 14);
      const thresholdKey = threshold.toISOString().slice(0, 10);

      for (const key of Object.keys(state.remindersDone)) {
        const datePart = key.split(":").pop();
        if (datePart < thresholdKey) {
          delete state.remindersDone[key];
        }
      }
    }

    function triggerReminder(title, item, key, note = "") {
      state.remindersDone[key] = true;
      beep();

      const message = note
        ? `${item}. ${note}.`
        : `${item}.`;

      showToast(title, message, true);
    }
